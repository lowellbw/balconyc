# One group for everyone. A crawler obeys only the most specific group that
# names it, so a separate "User-agent: GPTBot / Allow: /" group would silently
# drop the Disallow lines below for that bot. Listing every user agent in a
# single group (RFC 9309, section 2.2.1) keeps AI crawlers explicitly welcome
# and still out of /api/, /admin/ and /design/.

Disallow: /admin

# /design/ is the design workbench: mockups, forks and build scripts kept in
# the repo for reference. .vercelignore keeps it out of the deployment and
# vercel.json redirects any /design/ URL to the homepage; the Disallow above
# is a third guard.
